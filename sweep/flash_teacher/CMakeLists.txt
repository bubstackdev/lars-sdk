cmake_minimum_required(VERSION 3.24)
project(lars_flash_teacher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FLASH_REPO "/Users/ZKeown/Code/lab/flash" CACHE PATH "Read-only existing Flash checkout")
option(FLASH_BUILD_RUNTIME "Compile the actual Flash adapter as well as CPU tests" ON)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp HINTS
  "${CMAKE_CURRENT_SOURCE_DIR}/../../crates/lars-llama-sys/llama.cpp/vendor"
  "${FLASH_REPO}/third_party/llama.cpp-upstream/vendor" REQUIRED)
add_executable(flash-teacher-mock main.cpp)
target_include_directories(flash-teacher-mock PRIVATE "${JSON_INCLUDE_DIR}")
target_compile_definitions(flash-teacher-mock PRIVATE FLASH_CPU_ONLY=1)
if(FLASH_BUILD_RUNTIME)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  find_library(FLASH_LIBRARY NAMES flash PATHS "${FLASH_REPO}/build" NO_DEFAULT_PATH REQUIRED)
  find_library(LLAMA_LIBRARY NAMES llama PATHS "${FLASH_REPO}/third_party/llama.cpp-upstream/build/bin" NO_DEFAULT_PATH REQUIRED)
  add_executable(flash-teacher main.cpp runtime.mm)
  target_compile_options(flash-teacher PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_compile_definitions(flash-teacher PRIVATE FLASH_KERNEL_DIR="${FLASH_REPO}/kernels")
  target_include_directories(flash-teacher PRIVATE "${JSON_INCLUDE_DIR}" "${FLASH_REPO}/src" "${FLASH_REPO}/third_party/llama.cpp-upstream/include" "${FLASH_REPO}/third_party/llama.cpp-upstream/ggml/include")
  target_link_libraries(flash-teacher PRIVATE "${FLASH_LIBRARY}" "${LLAMA_LIBRARY}" "-framework Metal" "-framework Foundation")
  set_target_properties(flash-teacher PROPERTIES BUILD_RPATH "${FLASH_REPO}/third_party/llama.cpp-upstream/build/bin")
endif()
